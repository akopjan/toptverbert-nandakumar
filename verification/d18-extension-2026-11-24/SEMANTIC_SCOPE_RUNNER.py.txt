from pathlib import Path
import os
import subprocess
import time

core = Path('/workspace/scratch/9db1b532943d/akopyan-completion')
task = Path(__file__).resolve().parent
bin_dir = Path('/workspace/scratch/9db1b532943d/tools-runtime/lean-4.35.0-linux/bin')
env = os.environ.copy()
env['PATH'] = str(bin_dir) + os.pathsep + env.get('PATH', '')
env['MATHLIB_CACHE_DIR'] = '/workspace/scratch/9db1b532943d/tools-runtime/mathlib-cache'
env['LAKE_HOME'] = '/workspace/scratch/9db1b532943d/tools-runtime/lake-home'
paths_result = subprocess.run(
    [str(bin_dir / 'lake'), 'env', 'printenv', 'LEAN_PATH'],
    cwd=core, env=env, text=True, capture_output=True, check=True)
paths = [str((core / p).resolve()) if not Path(p).is_absolute() else p
         for p in paths_result.stdout.strip().split(os.pathsep)]
env['LEAN_PATH'] = os.pathsep.join([str(task), *paths])
for name in ['SemanticReference', 'SemanticScopeCheck']:
    started = time.monotonic()
    with (task / (name + '.log')).open('w') as log:
        result = subprocess.run(
            [str(bin_dir / 'lean'), '-j', '1', '-o', str(task / (name + '.olean')),
             str(task / (name + '.lean'))],
            cwd=task, env=env, stdout=log, stderr=subprocess.STDOUT)
    print(name, 'exit', result.returncode, 'seconds', round(time.monotonic() - started, 2),
          flush=True)
    if result.returncode:
        raise SystemExit(result.returncode)
