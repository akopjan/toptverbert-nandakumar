#!/usr/bin/env python3
"""Compare separately read, compiled Challenge and PalomarSolution interfaces.

This is an independent diagnostic, not the official Comparator, a sandboxed
workflow run, or a proof-body checker. No project source or object is written.
The .txt suffix keeps this evidence outside the project's script inventory.
"""
from __future__ import annotations

from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import re
import subprocess
import sys
import time

HERE = Path(__file__).resolve().parent
PROJECT = HERE.parent.parent
RUNTIME = PROJECT.parent / 'tools-runtime'
BIN = RUNTIME / 'lean-4.35.0-linux' / 'bin'
MODULES = ['Challenge', 'PalomarSolution']
NAMES = ['simplex', 'DisjointSupports', 'IsAlmostSixEmbedding', 'Ambient',
         'Body', 'unitBall', 'IsSixPartition', 'tverberg_counterexample',
         'convex_partition_counterexample_gt_17']
ENV = os.environ.copy()
ENV['PATH'] = str(BIN) + os.pathsep + ENV.get('PATH', '')
ENV['MATHLIB_CACHE_DIR'] = str(RUNTIME / 'mathlib-cache')
ENV['LAKE_HOME'] = str(RUNTIME / 'lake-home')


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def json_sha(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True,
        separators=(',', ':')).encode()).hexdigest()


def discover_project_sources():
    return {'.'.join(p.relative_to(PROJECT).with_suffix('').parts): p
            for p in PROJECT.rglob('*.lean')
            if not {'.lake', '.git', 'verification', 'tests'}.intersection(
                p.relative_to(PROJECT).parts[:-1])}


def project_import_closure(sources):
    """Bind the real project import closure, excluding unrelated Audit rebuilds."""
    pending = list(MODULES)
    found = set()
    while pending:
        module = pending.pop()
        if module in found:
            continue
        found.add(module)
        for line in sources[module].read_text().splitlines():
            match = re.match(r'^\s*(?:public\s+)?import\s+(?:all\s+)?(.+)$', line)
            if match:
                imports = match.group(1).split('--', 1)[0].split()
                pending.extend(m for m in imports if m in sources and m not in found)
    return sorted(found)


def bound_inputs(sources, closure):
    paths = [sources[m] for m in closure]
    paths += [PROJECT / name for name in
              ['lean-toolchain', 'lake-manifest.json', 'lakefile.toml']]
    paths += [HERE / 'serializer.lean.txt', Path(__file__).resolve()]
    for module in closure:
        stem = PROJECT / '.lake/build/lib/lean' / module.replace('.', '/')
        ordinary = Path(str(stem) + '.olean')
        if not ordinary.is_file():
            raise RuntimeError(f'Build the genuine module first: missing {ordinary}')
        paths.append(ordinary)
        for suffix in ['.olean.private', '.olean.server']:
            path = Path(str(stem) + suffix)
            if path.is_file():
                paths.append(path)
    return {str(p.relative_to(PROJECT)): sha(p) for p in sorted(set(paths))}


def save_result(result):
    (HERE / 'results.json').write_text(json.dumps(result, indent=2) + '\n')


def main():
    sources = discover_project_sources()
    closure = project_import_closure(sources)
    initial = bound_inputs(sources, closure)
    version_command = [str(BIN / 'lean'), '--version']
    version = subprocess.run(version_command, cwd=PROJECT, env=ENV,
        capture_output=True, text=True, check=True).stdout.strip()
    result = {
        'outcome': 'verification-in-progress',
        'started_utc': datetime.now(timezone.utc).isoformat(),
        'official_comparator': False,
        'official_sandbox_workflow': False,
        'theorem_proof_bodies_compared': False,
        'lean_version': version,
        'version_command': version_command,
        'driver_command': [sys.executable, str(Path(__file__).resolve())],
        'working_directory': str(PROJECT),
        'environment_overrides': {name: ENV[name] for name in
            ['PATH', 'MATHLIB_CACHE_DIR', 'LAKE_HOME']},
        'input_sha256_before': initial,
        'bound_project_import_closure': closure,
        'source_module_count': len(closure),
    }
    save_result(result)
    template = (HERE / 'serializer.lean.txt').read_text()
    stages = []
    for module in MODULES:
        source = HERE / ('Inspect' + module + '.lean')
        output = HERE / (module + '.json')
        source.write_text('module\npublic import Lean\nimport all ' + module + '\n' +
                          template.replace('__OUTPUT__', output.as_posix()))
        output.unlink(missing_ok=True)
        command = [str(BIN / 'lake'), 'env', 'lean', '-j', '1',
                   str(source.relative_to(PROJECT))]
        before = time.monotonic()
        with (HERE / (module + '.log')).open('w') as log:
            process = subprocess.run(command, cwd=PROJECT, env=ENV,
                                     stdout=log, stderr=subprocess.STDOUT)
        stage = {'module': module, 'command': command,
                 'working_directory': str(PROJECT),
                 'exit_code': process.returncode,
                 'seconds': round(time.monotonic() - before, 3),
                 'inspector_sha256': sha(source)}
        stages.append(stage)
        (HERE / (module + '.exit')).write_text(str(process.returncode) + '\n')
        if process.returncode != 0 or not output.is_file():
            result.update(outcome='inspection-failed', stages=stages,
                          input_sha256_after=bound_inputs(sources, closure))
            save_result(result)
            raise RuntimeError(f'{module} inspection failed; see its log')
    snapshots = {module: json.loads((HERE / (module + '.json')).read_text())
                 for module in MODULES}
    wanted = ['AkopyanSixfold.' + name for name in NAMES]
    for snapshot in snapshots.values():
        assert [d['name'] for d in snapshot['declarations']] == wanted
        assert [d['kind'] for d in snapshot['declarations']] == \
            ['definition'] * 7 + ['theorem'] * 2
    challenge = snapshots['Challenge']
    solution = snapshots['PalomarSolution']
    challenge_project_imports = sorted(set(challenge['imports']) & set(sources))
    solution_project_imports = sorted(set(solution['imports']) & set(sources))
    assert challenge_project_imports == ['Challenge']
    assert 'Solution' not in challenge['imports']
    assert 'PalomarSolution' not in challenge['imports']
    assert 'Challenge' not in solution['imports']
    assert 'Solution' in solution['imports']
    assert 'AkopyanResearch.Completion.DimensionEighteen' in solution['imports']
    assert set(challenge_project_imports + solution_project_imports) == set(closure), \
        'Static bound closure and actual imported project environments differ'
    differences = [name for name, a, b in zip(wanted, challenge['declarations'],
                                            solution['declarations']) if a != b]
    final = bound_inputs(sources, closure)
    unchanged = final == initial
    result.update(
        outcome=('compiled-interfaces-identical' if not differences else
                 'interfaces-differ') if unchanged else 'input-changed',
        finished_utc=datetime.now(timezone.utc).isoformat(), stages=stages,
        types_compared=9, definition_values_compared=7, differences=differences,
        challenge_project_imports=challenge_project_imports,
        proof_project_imports=solution_project_imports,
        solution_imports_challenge=False,
        excluded_expression_fields=['metadata', 'binder display names'],
        retained_expression_fields=['bound-variable indices', 'universe levels',
            'universe parameters', 'constant names', 'all applications',
            'instance arguments', 'binder types', 'binder implicitness',
            'let types and values', 'projections', 'literals'],
        input_sha256_after=final,
        input_bytes_unchanged=unchanged,
        serialized_sha256={m: sha(HERE / (m + '.json')) for m in MODULES},
        declarations_sha256={m: json_sha(snapshots[m]['declarations'])
                             for m in MODULES},
        declaration_comparisons=[{
            'name': name,
            'type_identical': a['type'] == b['type'],
            'universe_parameters_identical': a['level_params'] == b['level_params'],
            'definition_value_identical': a['value'] == b['value']
                if a['kind'] == 'definition' else None,
            'compiled_kind': a['kind'],
        } for name, a, b in zip(wanted, challenge['declarations'],
                                solution['declarations'])],
    )
    save_result(result)
    print(json.dumps({k: result[k] for k in ['outcome', 'types_compared',
        'definition_values_compared', 'differences', 'input_bytes_unchanged',
        'source_module_count', 'declarations_sha256']}, indent=2))
    return 1 if differences or not unchanged else 0


if __name__ == '__main__':
    sys.exit(main())
