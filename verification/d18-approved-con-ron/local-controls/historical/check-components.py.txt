#!/usr/bin/env python3
"""Exercise bundled checker components on our own, deliberately tiny fixture.

This is not a replacement for the official sandboxed Palomar check. In this
workspace that check fails before building because bwrap cannot mount its
temporary filesystems. No sandbox option is changed here. The supported
Comparator --*-from-export interface still performs the negative comparisons;
the bundled read-only checkers are additionally run as individual processes
inside the surrounding tool execution sandbox.
"""

import json
import os
from pathlib import Path
import subprocess


ROOT = Path(__file__).resolve().parent.parent
PREFIX = Path(subprocess.check_output(["lean", "--print-prefix"], text=True).strip())
AXIOMS = ["propext", "Classical.choice", "Quot.sound"]
TARGET = "Fixture.successorWitness"
PRIMITIVES = [
    "Nat.add", "Nat.sub", "Nat.mul", "Nat.pow", "Nat.gcd", "Nat.div", "Nat.mod",
    "Nat.beq", "Nat.ble", "Nat.land", "Nat.lor", "Nat.xor", "Nat.shiftLeft",
    "Nat.shiftRight", "String.ofList", "Char.ofNat", "List", "eagerReduce",
    "Nat", "String", "String.mk", "Char", "optParam", "autoParam",
    "semiOutParam", "outParam",
]
BUILTINS = ["Quot", "Quot.mk", "Quot.lift", "Quot.ind"]
CASES = {
    "positive": "SolutionPositive",
    "sorry": "SolutionSorry",
    "axiom": "SolutionAxiom",
    "hypothesis": "SolutionHypothesis",
}
RESULTS = []
for directory in ["exports", "configs", "logs", "tmp"]:
    (ROOT / directory).mkdir(exist_ok=True)
os.environ["TMPDIR"] = str(ROOT / "tmp")


def run(label, command, output=None):
    log = ROOT / "logs" / (label + ".log")
    if output:
        with output.open("wb") as out, log.open("wb") as err:
            result = subprocess.run(command, cwd=ROOT, stdout=out, stderr=err)
    else:
        with log.open("wb") as handle:
            result = subprocess.run(command, cwd=ROOT, stdout=handle, stderr=handle)
    RESULTS.append({"label": label, "command": command, "exit_code": result.returncode,
                    "log": str(log.relative_to(ROOT)),
                    "output": str(output.relative_to(ROOT)) if output else None})
    (ROOT / "component-results.json").write_text(json.dumps(RESULTS, indent=2) + "\n")
    print(f"{label}: exit {result.returncode}", flush=True)
    return result.returncode


for case, module in {"challenge": "Challenge", **CASES}.items():
    destination = ROOT / "exports" / (case + ".ndjson")
    status = run("export-" + case,
                 ["lake", "env", "leanexport", module, "--"]
                 + BUILTINS + [TARGET] + AXIOMS + PRIMITIVES,
                 destination)
    if status != 0:
        raise SystemExit(f"Exporter failed for {case}; inspect its log.")

for case, module in CASES.items():
    config = {
        "challenge_module": "Challenge",
        "solution_module": module,
        "theorem_names": [TARGET],
        "permitted_axioms": AXIOMS,
    }
    config_path = ROOT / "configs" / (case + ".json")
    config_path.write_text(json.dumps(config, indent=2) + "\n")
    run("comparator-export-" + case, [
        "lake", "comparator", "--config", str(config_path),
        "--challenge-from-export", str(ROOT / "exports" / "challenge.ndjson"),
        "--solution-from-export", str(ROOT / "exports" / (case + ".ndjson")),
    ])

for case in CASES:
    export_path = ROOT / "exports" / (case + ".ndjson")
    nanoda_config = {
        "use_stdin": False,
        "export_file_path": str(export_path),
        "permitted_axioms": AXIOMS,
        "unpermitted_axiom_hard_error": False,
        "num_threads": 4,
        "nat_extension": True,
        "string_extension": True,
    }
    config_path = ROOT / "configs" / ("nanoda-" + case + ".json")
    config_path.write_text(json.dumps(nanoda_config, indent=2) + "\n")
    run("leanchecker-" + case,
        [str(PREFIX / "bin" / "leanchecker"), "--silent", "--from-export", str(export_path)])
    run("nanoda-" + case, [str(PREFIX / "bin" / "nanoda_bin"), str(config_path)])
    run("con-ron-" + case, [str(PREFIX / "bin" / "con-ron"), str(export_path)])
