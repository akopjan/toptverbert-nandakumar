from pathlib import Path
from datetime import datetime, timezone
from collections import Counter
import hashlib
import json
import shutil
import subprocess
import time

ROOT = Path('/workspace/scratch/9db1b532943d')
MAIN = ROOT / 'akopyan-completion'
CLEAN = ROOT / 'clean-d18'
HERE = Path(__file__).resolve().parent
OUT = MAIN / 'verification/d18-clean-final/object-comparison'
LEAN = ROOT / 'tools-runtime/lean-4.35.0-linux/bin/lean'
SUFFIXES = ['.olean', '.olean.private', '.olean.server']


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def save(name, value):
    (OUT / name).write_text(json.dumps(value, indent=2) + '\n')


def input_snapshot(modules, sources):
    paths = sorted(set(sources) | {
        '.lake/build/lib/lean/' + m.replace('.', '/') + s
        for m in modules for s in SUFFIXES})
    return {str(root): {p: sha(root / p) for p in paths}
            for root in [MAIN, CLEAN]}


OUT.mkdir(parents=True, exist_ok=True)
main_gate = json.loads((MAIN / 'verification/d18-final-retry/results.json').read_text())
clean_gate = json.loads((MAIN / 'verification/d18-clean-final/results.json').read_text())
assert main_gate['input_sha256'] == clean_gate['input_sha256']
modules = main_gate['proof_modules']
assert len(modules) == 438
started = datetime.now(timezone.utc).isoformat()
before = input_snapshot(modules, main_gate['input_sha256'])
save('input-sha256-before.json', before)
for root in [MAIN, CLEAN]:
    for p, h in main_gate['input_sha256'].items():
        assert before[str(root)][p] == h, (root, p)

records = []
for module in modules:
    for suffix in SUFFIXES:
        path = '.lake/build/lib/lean/' + module.replace('.', '/') + suffix
        a, b = (MAIN / path).read_bytes(), (CLEAN / path).read_bytes()
        record = {'module': module, 'suffix': suffix, 'path': path,
                  'main_bytes': len(a), 'clean_bytes': len(b),
                  'main_sha256': hashlib.sha256(a).hexdigest(),
                  'clean_sha256': hashlib.sha256(b).hexdigest(),
                  'byte_identical': a == b}
        assert record['main_sha256'] == before[str(MAIN)][path]
        assert record['clean_sha256'] == before[str(CLEAN)][path]
        if suffix == '.olean.private' and a != b:
            offsets = [i for i, (x, y) in enumerate(zip(a, b)) if x != y]
            record['aligned_different_bytes'] = len(offsets)
            record['different_offsets_from_end'] = [len(a) - i for i in offsets]
        records.append(record)
save('object-hashes.json', records)
differing = sorted({r['module'] for r in records if not r['byte_identical']})
(OUT / 'differing-modules.txt').write_text('\n'.join(differing) + '\n')
shutil.copyfile(HERE / 'CompareModuleData.lean', OUT / 'CompareModuleData.lean.txt')
shutil.copyfile(Path(__file__), OUT / 'run_comparison.py.txt')
command = [str(LEAN), '--run', str(HERE / 'CompareModuleData.lean'),
           str(MAIN), str(CLEAN), str(OUT / 'differing-modules.txt'),
           str(OUT / 'constant-comparison.json')]
t = time.monotonic()
with (OUT / 'constant-comparison.log').open('w') as log:
    process = subprocess.run(command, cwd=ROOT, stdout=log, stderr=subprocess.STDOUT)
elapsed = time.monotonic() - t
(OUT / 'constant-comparison.exit').write_text(str(process.returncode) + '\n')
after = input_snapshot(modules, main_gate['input_sha256'])
save('input-sha256-after.json', after)
kernel = json.loads((MAIN / 'verification/d18-kernels/results.json').read_text())
kernel_input = kernel.get('input_sha256', {})
bound_objects = [r['path'] for r in records if r['path'] in kernel_input]
result = {
    'started_utc': started,
    'finished_utc': datetime.now(timezone.utc).isoformat(),
    'main_project': str(MAIN), 'clean_project': str(CLEAN),
    'proof_modules': modules, 'module_count': len(modules),
    'source_input_count': len(main_gate['input_sha256']),
    'source_inputs_match_successful_builds': True,
    'objects_compared': len(records),
    'byte_identical_objects': sum(r['byte_identical'] for r in records),
    'different_objects': sum(not r['byte_identical'] for r in records),
    'differing_modules': differing,
    'differences_by_suffix': dict(Counter(r['suffix'] for r in records
                                        if not r['byte_identical'])),
    'main_total_bytes': sum(r['main_bytes'] for r in records),
    'clean_total_bytes': sum(r['clean_bytes'] for r in records),
    'all_inputs_unchanged_during_comparison': before == after,
    'module_data_command': command, 'module_data_working_directory': str(ROOT),
    'module_data_exit_code': process.returncode,
    'module_data_seconds': round(elapsed, 3),
    'module_data_reader_sha256': sha(HERE / 'CompareModuleData.lean'),
    'lean_sha256': sha(LEAN),
    'kernel_record_outcome_at_snapshot': kernel.get('outcome'),
    'kernel_bound_object_count': len(bound_objects),
    'main_matches_kernel_bound_objects': all(before[str(MAIN)][p] == kernel_input[p]
                                              for p in bound_objects),
    'kernel_input_sha256_digest': hashlib.sha256(
        json.dumps(kernel_input, sort_keys=True, separators=(',', ':')).encode()).hexdigest(),
    'changes_to_project_sources_or_objects': False,
    'heavy_export_or_kernel_check_launched': False,
}
if process.returncode == 0:
    data = json.loads((OUT / 'constant-comparison.json').read_text())
    result['all_stored_nonextension_fields_equal'] = data['all_nonextension_fields_equal']
    result['outcome'] = ('same-declarations-different-metadata' if
        data['all_nonextension_fields_equal'] and before == after else 'comparison-difference')
else:
    result['outcome'] = 'module-data-inspection-failed'
save('results.json', result)
print(json.dumps({k: result[k] for k in ['outcome', 'module_count', 'objects_compared',
    'byte_identical_objects', 'different_objects', 'differences_by_suffix',
    'module_data_exit_code', 'all_inputs_unchanged_during_comparison',
    'kernel_bound_object_count', 'main_matches_kernel_bound_objects']}, indent=2), flush=True)
raise SystemExit(process.returncode or int(before != after))
